// פונקציית העוזר האישי המקומי (ללא צורך ב-API חיצוני) function handleAIAssistantQuery(userText) { const db = JSON.parse(localStorage.getItem('calorie_db_v1')) || { products: [], intake: {} }; const todayStr = new Date().toISOString().split('T')[0]; const todayIntake = db.intake[todayStr] || []; const text = userText.trim().toLowerCase(); // בדיקת בקשה לסיכום יומי if (text.includes('סיכום יומי') || text.includes('סיכום')) { let totalKcal = 0, totalProtein = 0, totalCarbs = 0, totalFat = 0; todayIntake.forEach(item => { totalKcal += (item.kcal || 0); totalProtein += (item.protein || 0); totalCarbs += (item.carbs || 0); totalFat += (item.fat || 0); }); const targetKcal = 2000; const diffKcal = targetKcal - totalKcal; let summaryMsg = `📊 סיכום יומי — ${todayStr}\n\n` + `🔥 נצרכו: ${Math.round(totalKcal)} קלוריות\n` + `🎯 יעד יומי: 2,000 קלוריות\n`; if (diffKcal >= 0) { summaryMsg += `🟢 נשארו: ${Math.round(diffKcal)} קלוריות\n`; } else { summaryMsg += `🔴 חריגה מהיעד: ${Math.abs(Math.round(diffKcal))} קלוריות\n`; } summaryMsg += `\n💪 חלבון: ${Math.round(totalProtein)} / 150 גרם\n` + `🍞 פחמימות: ${Math.round(totalCarbs)} / 200 גרם\n` + `🥑 שומן: ${Math.round(totalFat)} / 67 גרם`; return summaryMsg; } // זיהוי הוספת מזון מהמאגר (לדוגמה: "אכלתי 2 ביצים" או חיפוש מוצר) // העוזר סורק את המוצרים ב-db.products ומעדכן את ה-intake להיום let foundProduct = db.products.find(p => text.includes(p.name.toLowerCase())); if (foundProduct) { // חישוב יחסי לפי כמות או ברירת מחדל let multiplier = 1; // אפשר לחלץ מספרים מהטקסט אם צוין (למשל "2 יחידות" או "150 גרם") const matchNum = text.match(/(\d+)/); const amount = matchNum ? parseInt(matchNum[1]) : (foundProduct.unit_g || 100); if (text.includes('גרם')) { multiplier = amount / 100; } else if (text.includes('יחיד') || text.includes('ביצים') || text.includes('יח')) { multiplier = amount; // אם הכמות היא יחידות והחישוב הוא ליחידה או ל-100ג' } const consumedItem = { name: foundProduct.name, kcal: Math.round((foundProduct.kcal * (foundProduct.unit_g ? (foundProduct.unit_g/100) : 1)) * multiplier), protein: Math.round((foundProduct.protein || 0) * multiplier, 1), carbs: Math.round((foundProduct.carbs || 0) * multiplier, 1), fat: Math.round((foundProduct.fat || 0) * multiplier, 1), time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) }; if (!db.intake[todayStr]) db.intake[todayStr] = []; db.intake[todayStr].push(consumedItem); localStorage.setItem('calorie_db_v1', JSON.stringify(db)); return `רשמתי בהצלחה: ${consumedItem.name} (${consumedItem.kcal} קק"ל) לצריכה היומית!`; } return `הבנתי את הבקשה. תוכל לבקש ממני "סיכום יומי" או לכתוב למשל "אכלתי [שם מוצר מהמאגר]".`; }
