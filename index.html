//@version=6 strategy( "MNQ Trading Lab - Baseline V1.0-E1", overlay = true, pyramiding = 0, slippage = 1, calc_on_order_fills = false, calc_on_every_tick = false, process_orders_on_close = false, use_bar_magnifier = false ) //──────────────────────────────────── // CONSTANTS //──────────────────────────────────── string TZ = "America/New_York" float TICK = syminfo.mintick float TICK_VALUE = 0.50 float MAX_RISK = 125.0 float TARGET_R = 2.0 float DAILY_LOSS_LIMIT = 250.0 int MAX_TRADES_PER_DAY = 3 int MAX_CONTRACTS = 60 //──────────────────────────────────── // DAY STATE //──────────────────────────────────── int currentDay = year(time, TZ) * 10000 + month(time, TZ) * 100 + dayofmonth(time, TZ) int previousDay = year(time[1], TZ) * 10000 + month(time[1], TZ) * 100 + dayofmonth(time[1], TZ) bool newDay = currentDay != previousDay var int tradesToday = 0 var float dayStartNetProfit = 0.0 if barstate.isfirst dayStartNetProfit := strategy.netprofit if newDay tradesToday := 0 dayStartNetProfit := strategy.netprofit float dailyPnL = strategy.netprofit - dayStartNetProfit bool dailyLossHit = dailyPnL <= -DAILY_LOSS_LIMIT bool tradeLimitHit = tradesToday >= MAX_TRADES_PER_DAY //──────────────────────────────────── // OPENING RANGE // 09:30–09:45 ET //──────────────────────────────────── bool inOR = not na(time(timeframe.period, "0930-0945", TZ)) bool afterOR = not na(time(timeframe.period, "0945-1655", TZ)) var float orHigh = na var float orLow = na var bool orLocked = false if newDay orHigh := na orLow := na orLocked := false if inOR orHigh := na(orHigh) ? high : math.max(orHigh, high) orLow := na(orLow) ? low : math.min(orLow, low) if afterOR and not orLocked and not na(orHigh) and not na(orLow) orLocked := true //──────────────────────────────────── // STATE // 0 = FLAT // 1 = PENDING LONG // -1 = PENDING SHORT //──────────────────────────────────── var int state = 0 var float setupHigh = na var float setupLow = na var float entryPrice = na var float stopPrice = na var float targetPrice = na var float riskPerContract = na var int plannedContracts = 0 //──────────────────────────────────── // END OF DAY — EXECUTION FIX E1 //──────────────────────────────────── bool eod = not na(time(timeframe.period, "1655-1700", TZ)) if eod // Cancel pending entries strategy.cancel("Long Entry") strategy.cancel("Short Entry") // Force close any open position if strategy.position_size != 0 strategy.close_all(comment = "EOD") // Reset pending/setup state state := 0 setupHigh := na setupLow := na entryPrice := na stopPrice := na targetPrice := na riskPerContract := na plannedContracts := 0 //──────────────────────────────────── // BREAKOUT CONDITIONS // Only after OR is locked. //──────────────────────────────────── bool flat = strategy.position_size == 0 bool longBreakout = barstate.isconfirmed and orLocked and afterOR and close > orHigh bool shortBreakout = barstate.isconfirmed and orLocked and afterOR and close < orLow //──────────────────────────────────── // CREATE LONG SETUP //──────────────────────────────────── bool canCreateSetup = flat and state == 0 and not dailyLossHit and not tradeLimitHit and not eod if canCreateSetup and longBreakout setupHigh := high setupLow := low entryPrice := setupHigh + TICK stopPrice := setupLow - TICK float stopDistance = entryPrice - stopPrice float stopTicks = stopDistance / TICK riskPerContract := stopTicks * TICK_VALUE plannedContracts := riskPerContract > 0 ? math.floor(MAX_RISK / riskPerContract) : 0 plannedContracts := math.min(plannedContracts, MAX_CONTRACTS) targetPrice := entryPrice + stopDistance * TARGET_R if plannedContracts >= 1 state := 1 strategy.entry( "Long Entry", strategy.long, qty = plannedContracts, stop = entryPrice ) //──────────────────────────────────── // CREATE SHORT SETUP //──────────────────────────────────── if canCreateSetup and shortBreakout setupHigh := high setupLow := low entryPrice := setupLow - TICK stopPrice := setupHigh + TICK float stopDistance = stopPrice - entryPrice float stopTicks = stopDistance / TICK riskPerContract := stopTicks * TICK_VALUE plannedContracts := riskPerContract > 0 ? math.floor(MAX_RISK / riskPerContract) : 0 plannedContracts := math.min(plannedContracts, MAX_CONTRACTS) targetPrice := entryPrice - stopDistance * TARGET_R if plannedContracts >= 1 state := -1 strategy.entry( "Short Entry", strategy.short, qty = plannedContracts, stop = entryPrice ) //──────────────────────────────────── // OPPOSITE BREAKOUT CANCELS PENDING //──────────────────────────────────── if state == 1 and shortBreakout strategy.cancel("Long Entry") state := 0 if state == -1 and longBreakout strategy.cancel("Short Entry") state := 0 //──────────────────────────────────── // DETECT ACTUAL ENTRY FILL //──────────────────────────────────── bool longFilled = strategy.position_size > 0 and strategy.position_size[1] <= 0 bool shortFilled = strategy.position_size < 0 and strategy.position_size[1] >= 0 if longFilled or shortFilled tradesToday += 1 state := 0 //──────────────────────────────────── // EXIT ORDERS — ONLY WHILE IN POSITION //──────────────────────────────────── if strategy.position_size > 0 strategy.exit( "Long Exit", from_entry = "Long Entry", stop = stopPrice, limit = targetPrice ) if strategy.position_size < 0 strategy.exit( "Short Exit", from_entry = "Short Entry", stop = stopPrice, limit = targetPrice ) //──────────────────────────────────── // CANCEL ALL NEW ENTRIES AFTER LIMIT //──────────────────────────────────── if dailyLossHit or tradeLimitHit strategy.cancel("Long Entry") strategy.cancel("Short Entry") if strategy.position_size == 0 state := 0 //──────────────────────────────────── // VISUALS //──────────────────────────────────── plot(orHigh, "OR High", color = color.green) plot(orLow, "OR Low", color = color.red) plot( state != 0 ? entryPrice : na, "Pending Entry", color = color.blue, style = plot.style_linebr ) plot( strategy.position_size != 0 ? stopPrice : na, "Stop", color = color.red, style = plot.style_linebr ) plot( strategy.position_size != 0 ? targetPrice : na, "Target", color = color.green, style = plot.style_linebr )
